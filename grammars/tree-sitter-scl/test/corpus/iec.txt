TYPE E_State :
(
  Idle := 0,
  Running := 10,
  Stopping
) INT := Idle;
END_TYPE

VAR_GLOBAL
  bStart AT %I* : BOOL;
  nA, nB : INT;
END_VAR

INTERFACE I_Axis
  METHOD MoveTo : BOOL
  VAR_INPUT
    fTarget : REAL;
  END_VAR
  END_METHOD
END_INTERFACE

FUNCTION_BLOCK PUBLIC FB_Axis EXTENDS FB_Base IMPLEMENTS I_Axis
VAR
  eState : E_State;
  pCount : POINTER TO INT;
  rPos : REFERENCE TO REAL;
  sName : STRING(80);
  bLatch : BOOL;
END_VAR
CASE eState OF
  E_State.Idle: eState := E_State#Running;
  E_State.Running: eState := Stopping;
END_CASE
pCount^ := pCount^ + 1;
rPos REF= fPos;
bLatch S= bStart;
bLatch R= NOT bStart;
THIS^.nMoves := THIS^.nMoves + 1;
END_FUNCTION_BLOCK

METHOD PRIVATE Home : BOOL
Home := TRUE;
END_METHOD

PROGRAM MAIN
VAR
  fbAxis : FB_Axis;
END_VAR
fbAxis.MoveTo(fTarget := 1.5);
END_PROGRAM
